  function fuzzyHas(text, pat, k) {
    var m = pat.length, prev = new Array(m + 1), cur = new Array(m + 1), i, j;
    for (j = 0; j <= m; j++) prev[j] = j;
    if (prev[m] <= k) return true;
    for (i = 1; i <= text.length; i++) {
      cur[0] = 0;
      for (j = 1; j <= m; j++) {
        var cost = text.charAt(i - 1) === pat.charAt(j - 1) ? 0 : 1;
        cur[j] = Math.min(prev[j - 1] + cost, prev[j] + 1, cur[j - 1] + 1);
      }
      if (cur[m] <= k) return true;
      var t = prev; prev = cur; cur = t;
    }
    return false;
  }
  function scoreProduct(p, tokens) {
    var score = 0;
    for (var i = 0; i < tokens.length; i++) {
      var t = tokens[i];
      if (/^\d+$/.test(t)) {
        if (t === p._size) score += 10;
        else if (t.length >= 5 && p._code.indexOf(t) >= 0) score += 8;
        else if (p._hay.indexOf(t) >= 0) score += 4;
        else return 0;
      } else {
        if (p._hay.indexOf(t) >= 0 || p._code === t) score += 6;
        else if (t.length >= 3 && fuzzyHas(p._hay, t, t.length >= 7 ? 2 : 1)) score += 2;
        else return 0;
      }
    }
    return score;
  }

    var pool = S.products.slice();
    S.pl.items.forEach(function (it) { if (!S.byCode[it.code]) pool.push({ c: it.code, n: it.name, s: '', p: 0, _hay: norm(it.name), _code: it.code, _size: '' }); });
    var res = [];
    pool.forEach(function (p) {
      var sc = scoreProduct(p, tokens);
      if (sc > 0) res.push({ p: p, sc: sc + (inPL[p.c] ? 100 : 0) });
    });
    res.sort(function (a, b) { return b.sc - a.sc || String(a.p.n).localeCompare(String(b.p.n), 'th'); });
    return res.map(function (r) { return r.p; });
  }

  // ---------- count screen ----------
  function isCounted(code) { var c = S.counts[code]; return !!(c && (c.pack !== '' || c.unit !== '')); }
  function saveDraft() { if (S.pl) store('draft_' + S.pl.pl, { counts: S.counts, extra: S.extra }); }

  function renderList() {
    var list = currentList();
    var pages = Math.max(1, Math.ceil(list.length / PAGE_SIZE));
    if (S.page >= pages) S.page = pages - 1;
    if (S.page < 0) S.page = 0;
    var slice = list.slice(S.page * PAGE_SIZE, (S.page + 1) * PAGE_SIZE);
    var inPL = S.pl.inPL;
    var html = slice.map(function (p) {
      var c = S.counts[p.c] || { pack: '', unit: '' };
      var tags = '';
      if (p.s) tags += '<span class="tag size">' + esc(p.s) + '</span>';
      if (p.p) tags += '<span class="tag">' + esc(p.p) + ' ชิ้น/แพค</span>';
      tags += '<span class="tag">' + esc(p.c) + '</span>';
      if (!inPL[p.c]) tags += '<span class="tag out">⚠ นอก PL</span>';
      return '<div class="item' + (isCounted(p.c) ? ' done' : '') + '" data-code="' + esc(p.c) + '">' +
        '<div class="name">' + esc(p.n) + '</div><div class="tags">' + tags + '</div>' +
        '<div class="qty">' + stepper(p.c, 'pack', 'แพค', c.pack) + stepper(p.c, 'unit', 'ชิ้น', c.unit) + '</div></div>';
    }).join('');
    if (!list.length) html = '<div class="empty">ไม่พบสินค้าที่ค้นหา<br>ลองพิมพ์คำอื่น หรือขนาด เช่น 345</div>';
    $('list').innerHTML = html;
    $('pg-info').textContent = 'หน้า ' + (S.page + 1) + '/' + pages;
    $('pg-prev').disabled = S.page === 0;
    $('pg-next').disabled = S.page >= pages - 1;
    $('pager').classList.toggle('hidden', pages <= 1);
    $('q-clear').classList.toggle('hidden', !S.q && !S.focusCode);
    updateStat(list.length);
  }
  function stepper(code, field, label, val) {
    return '<div class="qbox"><div class="ql">' + label + '</div><div class="step">' +
      '<button type="button" data-act="minus" data-field="' + field + '">−</button>' +
      '<input type="text" inputmode="numeric" pattern="[0-9]*" data-field="' + field + '" value="' + esc(val) + '" placeholder="-">' +
      '<button type="button" data-act="plus" data-field="' + field + '">+</button></div></div>';
  }
  function updateStat(n) {
    var total = S.pl.items.length, done = 0;
    S.pl.items.forEach(function (it) { if (isCounted(it.code)) done++; });
    $('stat-left').innerHTML = (S.q || S.focusCode) ? 'ผลค้นหา <b>' + n + '</b> รายการ' : 'รายการใน PL <b>' + total + '</b>';
    $('stat-right').innerHTML = 'นับแล้ว <b>' + done + '/' + total + '</b>';
  }
  function setCount(code, field, val) {
    val = digits(val).replace(/^0+(?=\d)/, '');
    S.counts[code] = S.counts[code] || { pack: '', unit: '' };
    S.counts[code][field] = val;
    if (!S.pl.inPL[code] && S.extra.indexOf(code) < 0 && val !== '') S.extra.push(code);
    saveDraft();
    return val;
  }
  function onListInput(e) {
    var inp = e.target;
    if (inp.tagName !== 'INPUT') return;
    var card = inp.closest('.item');
    var code = card.getAttribute('data-code');
    var v = setCount(code, inp.getAttribute('data-field'), inp.value);
    if (inp.value !== v) inp.value = v;
    card.classList.toggle('done', isCounted(code));
    updateStat(currentList().length);
  }
  function onListClick(e) {
    var b = e.target.closest('button[data-act]');
    if (!b) return;
    var card = b.closest('.item'), code = card.getAttribute('data-code'), field = b.getAttribute('data-field');
    var inp = card.querySelector('input[data-field="' + field + '"]');
    var n = parseInt(inp.value || '0', 10);
    n = b.getAttribute('data-act') === 'plus' ? n + 1 : Math.max(0, n - 1);
    inp.value = setCount(code, field, String(n));
    card.classList.toggle('done', isCounted(code));
    updateStat(currentList().length);
  }

  // ---------- PL ----------
  function findPL(q, exact) {
    errBox('pl-err', ''); $('pl-multi').innerHTML = '';
    var d = digits(q);
    if (!exact && d.length < 5) { errBox('pl-err', 'กรุณาพิมพ์เลข PL อย่างน้อย 5 หลัก'); return; }
    loading(true, 'กำลังค้นหา PL...');
    api('findPL', { q: q, exact: !!exact }).then(function (j) {
      if (j.multiple) {
        $('pl-multi').innerHTML = '<p class="sec" style="margin-top:16px">พบหลายใบ กรุณาเลือก:</p>' +
          j.multiple.map(function (m) { return '<button class="choice" data-pl="' + esc(m.pl) + '">' + esc(m.pl) + ' <span class="muted">(' + m.items + ' รายการ)</span></button>'; }).join('');
        return;
      }
      openPL(j);
    }).catch(function (e) { errBox('pl-err', e.message); })
      .finally(function () { loading(false); });
  }
  function openPL(j) {
    S.pl = { pl: j.pl, items: j.items, inPL: {} };
    j.items.forEach(function (it) { S.pl.inPL[it.code] = true; });
    S.counts = {}; S.extra = [];
    Object.keys(j.saved || {}).forEach(function (code) {
      var s = j.saved[code];
      var bothZero = !s.pack && !s.unit;
      S.counts[code] = { pack: (s.pack || bothZero) ? String(s.pack) : '', unit: s.unit ? String(s.unit) : '' };
      if (!S.pl.inPL[code]) S.extra.push(code);
    });
    var draft = store('draft_' + j.pl);
    if (draft && draft.counts) { S.counts = draft.counts; S.extra = draft.extra || []; }
    S.page = 0; S.q = ''; S.focusCode = null; $('q').value = '';
    setText('pl-no', j.pl);
    show('s-count');
    renderList();
    if (j.lastBy) toast('PL นี้เคยบันทึกโดย ' + j.lastBy + ' (' + j.lastAt + ') — แก้ไขได้', '');
  }

  // ---------- review & submit ----------
  function renderReview() {
    var counted = [], missing = [], extra = [];
    S.pl.items.forEach(function (it) {
      var p = S.byCode[it.code] || { c: it.code, n: it.name };
      if (isCounted(it.code)) counted.push(p); else missing.push(p);
    });
    S.extra.forEach(function (c) { if (isCounted(c) && !S.pl.inPL[c]) extra.push(S.byCode[c] || { c: c, n: c }); });
    function qtyText(code) {
      var c = S.counts[code], t = [];
      if (c.pack !== '') t.push(c.pack + ' แพค');
      if (c.unit !== '') t.push(c.unit + ' ชิ้น');
      return t.join(' + ');
    }
    function rows(list, withQty) {
      return list.map(function (p) { return '<div class="rv"><span class="n">' + esc(p.n) + (p.s ? ' <span class="tag size">' + esc(p.s) + '</span>' : '') + '</span>' + (withQty ? '<span class="q">' + esc(qtyText(p.c)) + '</span>' : '') + '</div>'; }).join('');
    }
    var h = '';
    if (missing.length) h += '<div class="card" style="border:3px solid var(--red)"><div class="sec red">⚠️ ยังไม่ได้นับ ' + missing.length + ' รายการ</div>' + rows(missing, false) + '</div>';
    h += '<div class="card"><div class="sec green">✔ นับแล้ว ' + counted.length + ' รายการ</div>' + (rows(counted, true) || '<div class="muted">-</div>') + '</div>';
    if (extra.length) h += '<div class="card"><div class="sec orange">➕ สินค้านอก PL ' + extra.length + ' รายการ</div>' + rows(extra, true) + '</div>';
    $('review-body').innerHTML = h;
  }
  function submit() {
    var items = [];
    Object.keys(S.counts).forEach(function (code) {
      if (!isCounted(code)) return;
      var c = S.counts[code];
      items.push({ code: code, pack: c.pack === '' ? 0 : +c.pack, unit: c.unit === '' ? 0 : +c.unit });
    });
    if (!items.length) { toast('ยังไม่ได้ใส่จำนวนสินค้าเลย', 'bad'); return; }
    loading(true, 'กำลังส่งข้อมูล...');
    api('submit', { pl: S.pl.pl, items: items }).then(function (j) {
      store('draft_' + S.pl.pl, null);
      $('done-msg').textContent = 'บันทึก ' + j.saved + ' รายการ • ' + j.time;
      show('s-done');
    }).catch(function (e) { toast('ส่งไม่สำเร็จ: ' + e.message, 'bad'); })
      .finally(function () { loading(false); });
  }

  // ---------- scanner ----------
  var scanner = null, scanning = false;
  function openScanner() {
    if (!window.Html5Qrcode) { toast('โหลดตัวสแกนไม่ได้ ตรวจสอบอินเทอร์เน็ต', 'bad'); return; }
    $('scanner').classList.remove('hidden');
    var F = Html5QrcodeSupportedFormats;
    scanner = new Html5Qrcode('reader', {
      formatsToSupport: [F.EAN_13, F.EAN_8, F.UPC_A, F.UPC_E, F.CODE_128, F.CODE_39, F.ITF],
      experimentalFeatures: { useBarCodeDetectorIfSupported: true }, verbose: false
    });
    scanning = true;
    scanner.start({ facingMode: 'environment' },
      { fps: 12, qrbox: function (w, h) { return { width: Math.floor(w * 0.85), height: Math.floor(Math.min(h, w) * 0.45) }; } },
      onScan, function () {}
    ).catch(function (e) {
      closeScanner();
      toast('เปิดกล้องไม่ได้ กรุณาอนุญาตการใช้กล้อง', 'bad');
    });
  }
  function closeScanner() {
    $('scanner').classList.add('hidden');
    if (scanner && scanning) { scanning = false; scanner.stop().then(function () { scanner.clear(); }).catch(function () {}); }
  }
  function onScan(text) {
    if (!scanning) return;
    closeScanner();
    if (navigator.vibrate) navigator.vibrate(120);
    handleBarcode(String(text).trim());
  }
  function handleBarcode(code) {
    var hit = S.byBarcode[code] || S.byBarcode[code.replace(/^0+/, '')] || (S.byCode[code] ? { code: code, field: 'pack' } : null);
    if (!hit) { toast('ไม่พบบาร์โค้ด ' + code + ' ในระบบ', 'bad'); return; }
    S.focusCode = hit.code; S.q = ''; $('q').value = ''; S.page = 0;
    renderList();
    var p = S.byCode[hit.code];
    toast('พบ: ' + (p ? p.n : hit.code) + (S.pl.inPL[hit.code] ? '' : ' (นอก PL)'), S.pl.inPL[hit.code] ? 'ok' : 'bad');
    var card = document.querySelector('.item');
    if (card) {
      card.classList.add('flash');
      var inp = card.querySelector('input[data-field="' + hit.field + '"]');
      inp.classList.add('hl');
      setTimeout(function () { inp.focus(); }, 150);
    }
  }

  // ---------- start ----------
  updateDates();

  var u = store('user');

  refreshProducts(true);

  // สำหรับทดสอบ
  window.__plc = { S: S, handleBarcode: handleBarcode, renderList: renderList };
})();
